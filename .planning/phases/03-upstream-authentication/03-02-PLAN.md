type HeaderStrategy struct {
    name  string
    value string
}

func NewHeaderStrategy(cfg *HeaderConfig) (*HeaderStrategy, error) {
    // Expand environment variables with validation
    value, err := config.ExpandEnvWithValidation(cfg.Value)
    if err != nil {
        return nil, fmt.Errorf("header auth value: %w", err)
    }
    return &HeaderStrategy{name: cfg.Name, value: value}, nil
}

func (s *HeaderStrategy) RoundTripper(base http.RoundTripper) http.RoundTripper {
    return &headerRoundTripper{base: base, name: s.name, value: s.value}
}

type headerRoundTripper struct {
    base  http.RoundTripper
    name  string
    value string
}

func (rt *headerRoundTripper) RoundTrip(req *http.Request) (*http.Response, error) {
    // CRITICAL: Clone request to avoid modifying original (per RESEARCH.md Pitfall 3)
    reqCopy := req.Clone(req.Context())
    reqCopy.Header.Set(rt.name, rt.value)
    return rt.base.RoundTrip(reqCopy)
}
```

type BasicStrategy struct {
    username string
    password string
}

func NewBasicStrategy(cfg *BasicConfig) (*BasicStrategy, error) {
    // Expand environment variables with validation
    username, err := config.ExpandEnvWithValidation(cfg.Username)
    if err != nil {
        return nil, fmt.Errorf("basic auth username: %w", err)
    }
    password, err := config.ExpandEnvWithValidation(cfg.Password)
    if err != nil {
        return nil, fmt.Errorf("basic auth password: %w", err)
    }
    return &BasicStrategy{username: username, password: password}, nil
}

func (s *BasicStrategy) RoundTripper(base http.RoundTripper) http.RoundTripper {
    return &basicRoundTripper{base: base, username: s.username, password: s.password}
}

type basicRoundTripper struct {
    base     http.RoundTripper
    username string
    password string
}

func (rt *basicRoundTripper) RoundTrip(req *http.Request) (*http.Response, error) {
    // CRITICAL: Clone request to avoid modifying original
    reqCopy := req.Clone(req.Context())
    // Use stdlib SetBasicAuth per RESEARCH.md "Don't Hand-Roll" section
    reqCopy.SetBasicAuth(rt.username, rt.password)
    return rt.base.RoundTrip(reqCopy)
}
```

Test cases:

- Environment variables expanded for both username and password
- Original request unmodified
