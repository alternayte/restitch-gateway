                    Status:      200,
                    Body:        map[string]interface{}{"result": "ok"},
                    ContentType: "application/json",
                },
            },
        },
    }

    _, err := CompileConfig(cfg)
    if err == nil {
        t.Fatal("expected error for missing step reference, got nil")
    }
    if !contains(err.Error(), "non-existent") && !contains(err.Error(), "not found") {
        t.Errorf("expected 'non-existent' or 'not found' error, got: %v", err)
    }
}

func contains(s, substr string) bool {
    return len(s) >= len(substr) && (s == substr || len(s) > len(substr) && findSubstring(s, substr))
}

func findSubstring(s, substr string) bool {
    for i := 0; i <= len(s)-len(substr); i++ {
        if s[i:i+len(substr)] == substr {
            return true
        }
    }
    return false
}
